private String computeSHA512(String data) throws Exception {
MessageDigest digest = MessageDigest.getInstance(SHA_512);
byte[] hash = digest.digest(data.getBytes(StandardCharsets.UTF_8)); // Use UTF-8
encoding
StringBuilder sb = new StringBuilder();
for (byte b : hash) {
sb.append(String.format(FORMAT, b));
}
